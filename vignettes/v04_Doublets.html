Metadata cleanup and annotation

This chunk (currently not executed) shows how: - Sample names are assigned to each object - Redundant metadata is removed - Conditions and genotypes are annotated based on sample.id - Prediction scores are merged from an external table (pred)

Code 
names(scelist) <- lapply(scelist, function(sce){unique(sce$sample.id)})
scelist <- lapply(scelist, function(sce)
{
    colData(sce) <- colData(sce)[, -which(colnames(colData(sce)) %in% 
                                    c("SingleR", "Azimuth", "Azimuth_filteredCells", "SingleR_filteredCells"))]
    colData(sce) <- colData(sce)[, -13] ## 13 is a duplicated barcode col
    sce$condition <- ifelse( length(grep("SD", sce$sample.id))!=0, "SD", "CTRL")
    sce$genotype <- ifelse( length(grep("S3", sce$sample.id))!=0, "S3", "WT")
    
    sce$replicate <- strsplit(x=sce$sample.id[1], "_")[[1]][1]
    sce
})

scelist <- lapply(scelist, function(sce) {sce[,sce$in_signac]})
scelist <- lapply(scelist, function(sce)
{
    cidx <- grep(paste0(unique(sce$sample.id),"$"), rownames(pred))
    predsce <- pred[cidx,]
    rownames(predsce) <- gsub(paste0("_", unique(sce$sample.id)), "", rownames(predsce))
    idx <- match(colnames(sce), rownames(predsce))
    colData(sce) <- cbind(colData(sce), predsce[idx,])
    colnames(colData(sce)) <- gsub("scores", "SRscores", colnames(colData(sce)))
    sce
})

Summary Table

Generate summary table of cell type counts

This chunk builds a summary table that compares the number of cells per label before and after filtering, across all samples. It: - Constructs a matrix of cell type counts per sample - Matches labels across filtered and unfiltered data - Displays the tables using DT::datatable() for interactive visualization

Code 
nms <- unlist(lapply(scelist, function(sce)
{
    unique(paste0(sce$replicate,  "_", sce$genotype, "_", sce$condition ))
}))

names(nms) <- NULL
lbls <- sort(unique(unlist(lapply(seq_along(scelist[idxordlist]), function(i)
{
    unique(scelist[[i]]$labels)
}))))

scelist <- scelist[idxordlist]
scelistf <- scelistf[idxordlist]
scelistdo <- scelistdo[idxordlist]
scelistdof <- scelistdof[idxordlist]
nms <- nms[idxordlist]

df <- data.frame(labels=lbls)
for(i in seq_along(scelist))
{
    # print(i)
    sn <- nms[i]
    tbls <- table(scelist[[i]]$labels)
    nc <- rep(0, dim(df)[1])
    names(nc) <- df[,1]
    nc[match(names(tbls), names(nc))] <- tbls
    nmdf <- colnames(df)
    df <- cbind.data.frame(df, nc)
    tblsf <- table(scelistf[[i]]$labels)
    nc <- rep(NA, dim(df)[1])
    names(nc) <- df[,1]
    nc[match(names(tblsf), names(nc))] <- tblsf
    df <- cbind.data.frame(df, nc)
    colnames(df) <- c(nmdf, sn, paste0(sn, "_filt"))
}
# df
df <- df[,-1]
dflist <- list()
j=1
for (i in 1:dim(df)[2])
{
    if ((i %% 4) == 0)
    {
        dflist[[j]] <- df[,c((i-3):i)]
        j=j+1
    }
}
