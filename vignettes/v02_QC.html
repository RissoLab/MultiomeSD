Number of cells by metrics and cell types

Here we break down the same QC metrics, but stratified by cell type within each sample, to see how filtering affects individual subclusters.

Code 
for (i in 1:12)
{
    sc <- scelist[[i]]
    idxct <- which(unique(sc$subcluster) %in% ct)
    sf <- matrix(nrow=length(idxct), ncol=7)
    k <- 1
    for (j in seq_along(ct))
    {
        idx <- which(sc$subcluster==ct[j])
        if (length(idx)==0)
        {
            print(paste0(names(scelist)[i], " has not any cells for ", ct[j], " cell type\n"))
            next
        }
        sce <- sc[,idx]
        tot <- dim(sce)[2]
        natac <- sum(sce$nCount_ATAC_passed)
        nrna <- sum(sce$nCount_RNA_passed)
        nfea <- sum(sce$nFeature_RNA_passed)
        nnucl <- sum(sce$nucleosome_passed)
        ntss <- sum(sce$TSS_passed)
        nsign <- sum(sce$in_signac)
        sf[k,] <- c(tot, natac, nrna, nfea, nnucl, ntss, nsign)
        k <- k+1
    }
    rownames(sf) <- ct[idxct]
    colnames(sf) <- c("Starting", "ATAC_s", "RNA_s", "Feature_s", "Nucleosome_s", "TSS_s", "Signac")
    sfd <- as.data.frame(sf)
    print(paste0("Sample ", names(scelist)[i], " Cell Type ", ct[j],"\n"))
    print(knitr::kable(sfd))
    cat("\n")
}

Convert logical metadata columns

This section converts all logical (TRUE/FALSE) metadata columns into string labels (“1. TRUE” / “2. FALSE”) for easier tabulation and plotting.

Code 
scelist <- lapply(scelist, function(sce) {
    for(i in 1:dim(colData(sce))[2])
    {
        if( is.logical(colData(sce)[,i]) )
        {
            colData(sce)[,i] <- ifelse(colData(sce)[,i]==TRUE, "1. TRUE", "2. FALSE")
        }
    }
    return(sce)
})

Filtered vs unfiltered cells

{
    sce <- scelist[[i]]

    print(plotFilteredCells(sce, filterCellsBy="custom", customColName="in_signac", name=paste0(names(scelist)[i], " signac_filtered")))
})
